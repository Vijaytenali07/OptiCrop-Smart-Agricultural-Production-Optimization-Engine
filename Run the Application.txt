from flask import Flask, request, render_template_string
import pandas as pd
import joblib

# ----------------------------------------------------------
# Initialize Flask App
# ----------------------------------------------------------
app = Flask(__name__)

# ----------------------------------------------------------
# Load Model
# ----------------------------------------------------------
model = joblib.load("Best_OptiCrop_Model.pkl")
scaler = joblib.load("OptiCrop_Scaler.pkl")
label_encoder = joblib.load("OptiCrop_LabelEncoder.pkl")

# ----------------------------------------------------------
# HTML Template
# ----------------------------------------------------------
HTML = """
<!DOCTYPE html>
<html>

<head>

<title>OptiCrop</title>

<style>

body{

background:#eef7ee;
}

.container{

}

h1{

color:green;
}

label{
font-weight:bold;
}

input{
width:100%;
padding:10px;
margin-top:5px;
margin-bottom:15px;
}

button{
width:100%;
padding:12px;
background:green;
color:white;
border:none;

cursor:pointer;
}

.result{
margin-top:20px;
padding:15px;
background:#dff0d8;
text-align:center;

font-weight:bold;
}

</style>

</head>

<body>

<div class="container">

<h1>🌱 OptiCrop</h1>

<form method="POST">

<label>Nitrogen</label>
<input type="number" step="0.01" name="Nitrogen" required>

<label>Phosphorus</label>
<input type="number" step="0.01" name="Phosphorus" required>

<label>Potassium</label>
<input type="number" step="0.01" name="Potassium" required>

<label>Temperature</label>
<input type="number" step="0.01" name="Temperature" required>

<label>Humidity</label>
<input type="number" step="0.01" name="Humidity" required>

<label>Soil pH</label>
<input type="number" step="0.01" name="pH" required>

<label>Rainfall</label>
<input type="number" step="0.01" name="Rainfall" required>

<button type="submit">

Predict Best Crop

</button>

</form>

{% if prediction %}

<div class="result">

Recommended Crop:<br><br>

{{ prediction }}

</div>

{% endif %}

</div>

</body>

</html>
"""

# ----------------------------------------------------------
# Home Route
# ----------------------------------------------------------
@app.route("/", methods=["GET", "POST"])
def home():

    prediction = None

    if request.method == "POST":

        nitrogen = float(request.form["Nitrogen"])
        phosphorus = float(request.form["Phosphorus"])
        potassium = float(request.form["Potassium"])
        temperature = float(request.form["Temperature"])
        humidity = float(request.form["Humidity"])
        soil_ph = float(request.form["pH"])
        rainfall = float(request.form["Rainfall"])

        sample = pd.DataFrame([{
            "Nitrogen": nitrogen,
            "Phosphorus": phosphorus,
            "Potassium": potassium,
            "Temperature": temperature,
            "Humidity": humidity,
            "pH": soil_ph,
            "Rainfall": rainfall
        }])

        sample_scaled = scaler.transform(sample)

        pred = model.predict(sample_scaled)

        prediction = label_encoder.inverse_transform(pred)[0]

    return render_template_string(
        HTML,
        prediction=prediction
    )

# ----------------------------------------------------------
# Run Flask App
# ----------------------------------------------------------
if __name__ == "__main__":

    print("=" * 60)
    print("OptiCrop Server Started Successfully")
    print("Open Browser:")
    print("http://127.0.0.1:5000")
    print("=" * 60)

    app.run(debug=True)
